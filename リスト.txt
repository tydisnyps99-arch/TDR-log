// Webアプリからのデータ受け取り（保存処理）
function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheetName = data.sheetName || 'logs'; // 指定がなければ logs シート
    const sheet = ss.getSheetByName(sheetName);

    if (!sheet) {
      throw new Error('指定されたシートが存在しません: ' + sheetName);
    }

    // 1行目の見出し一覧を取得
    const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
    
    // 見出しに合わせてデータの並び順を自動調整
    const row = headers.map(header => {
      if (header === '日時' || header === 'timestamp') {
        return new Date();
      }
      return data.payload[header] !== undefined ? data.payload[header] : '';
    });

    sheet.appendRow(row);

    return ContentService.createTextOutput(JSON.stringify({ status: 'success' }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: error.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// Webアプリへのデータ受け渡し（読み込み処理）
function doGet(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheetName = e.parameter.sheetName || 'logs';
    const sheet = ss.getSheetByName(sheetName);

    if (!sheet) {
      throw new Error('指定されたシートが存在しません: ' + sheetName);
    }

    const data = sheet.getDataRange().getValues();
    if (data.length <= 1) {
      return ContentService.createTextOutput(JSON.stringify({ status: 'success', data: [] }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    const headers = data[0];
    const rows = data.slice(1);

    // オブジェクト配列に変換して返す
    const result = rows.map(row => {
      let obj = {};
      headers.forEach((header, index) => {
        obj[header] = row[index];
      });
      return obj;
    });

    return ContentService.createTextOutput(JSON.stringify({ status: 'success', data: result }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: error.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

/**
 * 施設マスターデータをスプレッドシートの「マスターデータ」シートに出力する関数
 */
function exportMasterToSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheetName = 'マスターデータ';
  let sheet = ss.getSheetByName(sheetName);
  
  // シートが存在しない場合は新規作成
  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
  } else {
    // 既存のシートがある場合は一旦クリア
    sheet.clear();
  }
  
  // 1行目のヘッダー
  const headers = [
    "ID", "パーク", "エリア", "カテゴリ", "施設名", 
    "DPA価格", "スタンバイパス", "モバイルオーダー", "エントリー受付", "タイミングタイプ"
  ];
  
  // オブジェクト配列からセル用の二次元配列に変換
  const rows = facilityMaster.map(item => [
    item.id,
    item.park,
    item.area,
    item.category,
    item.name,
    item.dpaPrice !== null ? item.dpaPrice : "-",
    item.standbyPass ? "対象" : "対象外",
    item.mobileOrder ? "対象" : "対象外",
    item.entryReception ? "対象" : "対象外",
    item.timingType || "-"
  ]);
  
  // ヘッダーとデータを一括でシートに書き込み
  sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
  sheet.getRange(2, 1, rows.length, headers.length).setValues(rows);
  
  // 見栄えの調整（1行目を太字＆背景色変更、列幅の自動調整）
  sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#e6f2ff");
  sheet.setFrozenRows(1);
  for (let i = 1; i <= headers.length; i++) {
    sheet.autoResizeColumn(i);
  }
  
  Logger.log("「マスターデータ」シートへの出力が完了しました！");
}
